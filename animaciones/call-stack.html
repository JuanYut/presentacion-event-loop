Código
function formatear(nombre) {
  return nombre.toUpperCase();
}

function saludar(nombre) {
  const texto = formatear(nombre);
  console.log("Hola, " + texto);
}

function main() {
  console.log("Inicio");
  saludar("Ana");
  console.log("Fin");
}

main();
